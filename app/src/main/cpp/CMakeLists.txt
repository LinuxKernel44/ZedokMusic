cmake_minimum_required(VERSION 3.22)
project(zedokaudio C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ZK_COMMON ${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/common)
find_package(oboe REQUIRED CONFIG)
file(GLOB ZSTD_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/common/*.c ${CMAKE_CURRENT_SOURCE_DIR}/third_party/zstd/decompress/*.c)
add_library(zstd_dec STATIC ${ZSTD_SRC})
target_compile_definitions(zstd_dec PRIVATE ZSTD_DISABLE_ASM=1 ZSTD_LEGACY_SUPPORT=0 DYNAMIC_BMI2=0)
set_target_properties(zstd_dec PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99)
# LAME (LGPL) as its own shared library, linked dynamically (see third_party/lame/README.zedok).
file(GLOB LAME_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lame/libmp3lame/*.c)
list(FILTER LAME_SRC EXCLUDE REGEX "mpglib_interface")
add_library(mp3lame SHARED ${LAME_SRC})
target_include_directories(mp3lame PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lame/include ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lame/libmp3lame)
target_compile_definitions(mp3lame PRIVATE ieee754_float32_t=float SIZEOF_SHORT=2 SIZEOF_INT=4 SIZEOF_LONG=8 SIZEOF_LONG_LONG=8 SIZEOF_FLOAT=4 SIZEOF_DOUBLE=8 SIZEOF_LONG_DOUBLE=16 STDC_HEADERS=1 HAVE_STDINT_H=1 HAVE_LIMITS_H=1 HAVE_ERRNO_H=1 HAVE_STRING_H=1 HAVE_STRCHR=1 HAVE_MEMCPY=1 HAVE_INTTYPES_H=1 HAVE_UNISTD_H=1 NDEBUG=1 LAME_LIBRARY_BUILD=1)
target_compile_options(mp3lame PRIVATE -O2 -w)
set_target_properties(mp3lame PROPERTIES C_STANDARD 99)
target_link_options(mp3lame PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(mp3lame m)
add_library(zedokaudio SHARED jni.cpp engine.cpp wav.cpp extract.cpp recorder.cpp xclient.cpp flac_enc.cpp mp3_enc.cpp)
target_compile_options(zedokaudio PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter)
target_include_directories(zedokaudio PRIVATE ${ZK_COMMON} ${CMAKE_CURRENT_SOURCE_DIR}/third_party/lame/include)
# Android 15+ devices may use 16 KB memory pages: native libraries must be aligned for them.
target_link_options(zedokaudio PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(zedokaudio mp3lame zstd_dec oboe::oboe log android jnigraphics)
